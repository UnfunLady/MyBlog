### 笔记描述

> 用于录制音频上传解析

### 主要代码

```js
$global.isRecording = false;
$global.audioChunks = [];

开始录音方法: async () => {
  try {
    const stream = await navigator.mediaDevices.getUserMedia({
      audio: true,
    });
    $global.mediaRecorder = new MediaRecorder(stream);
    $global.mediaRecorder.ondataavailable = (event) => {
      $global.audioChunks.push(event.data);
    };
    $global.mediaRecorder.onstop = () => {
      const audioBlob = new Blob($global.audioChunks, { type: "audio/wav" });
      console.log(audioBlob);
      // 上传
      // uploadAudio(audioBlob);
    };
    $global.mediaRecorder.start();
    $global.isRecording = true;
  } catch (e) {
    console.log(e);
  }
};

停止录音方法: () => {
  try {
    if (!$global.isRecording) {
      return;
    }
    if ($global.mediaRecorder) {
      $global.mediaRecorder.stop();
      $global.isRecording = false;
    }
  } catch (e) {
    console.log(e);
  }
};
```
