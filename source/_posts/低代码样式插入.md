### 笔记描述

> 当低代码组件样式无法满足需求时候用这个插入样式表

### 主要代码

```js
$effect(() => {
  $props({
    className: "custom-modal1",
  });
  const style = document.createElement("style");
  style.innerHTML = `
   .custom-card-head-title {
     margin-left:20px
   }
   .custom-modal-header {
     background:#f4f4f4 !important
   }
   .custom-modal-content {
    background:#f4f4f4 !important
   }

   .custom-modal1 .custom-modal-body{
     max-height:85vh !important;
     overflow:auto !important;
   }

   .custom-list-call{
     display:flex !important;
     overflow-x:auto !important;
     gap:30px;
     padding-bottom:10px;
   }

  .custom-list-call .adm-pull-to-refresh-content{
    display:flex !important;
    flex-wrap:noWrap !important;
  }
   .custom-list-call-item{
    flex-basis:auto !important;
   }

  .custom-list-call::-webkit-scrollbar {
    width: 4px; /* 滚动条的宽度 */
    height: 4px; /* 滚动条的高度 */
}
 
  .custom-list-call::-webkit-scrollbar-thumb {
    background-color: rgba(0, 0, 0, 0.08); 
    border-radius: 10px; 
}
 
  .custom-list-call::-webkit-scrollbar-track {
    background: #f4f4f4; 
}
  
  `;
  document.head.appendChild(style);
  return () => {
    style.remove();
  };
}, []);
```
